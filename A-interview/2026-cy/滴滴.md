const arr = [1, 2, 5, 4] 
function aSort(arr) {
    if (arr.length <= 1) {
        return arr;
    }
    const left = [];
    const right = [];
    const len = arr.length;
    const temp = arr[Math.floor(len / 2)-1]; // 取整
    arr.forEach(item => {
        if (item < temp) {
            left.push(item);
        } else if (item > temp) {
            right.push(item);
        }
    })

    return [...aSort(left), temp, ...aSort(right)];
}

aSort(arr); // [1, 2, 4, 5]

function classNames(...args) {
    // 输入数组： 字符串、对象、数组（嵌套）
    const result = [];
    if (Array.isArray(args) && args.length > 0) {
        args.forEach(item => {
            // 0. null undefined ''
            if (!item) {
                return
            }
            // 1. 纯字符串
            if (typeof item === 'string') {
                result.push(item);
            }

            // 3. 数组 ['b', { c: true, d: false }]
            if (Array.isArray(item)) {
                // 拍平
                for (let v of item) {
                    debugger
                    result.push(classNames(v))
                }

            } else if (item instanceof Object) {
                // 2. 对象 { bar: true }
                for (let key in item) {
                    // 如果值为true，加入result
                    debugger
                    if (item[key]) {
                        result.push(key);
                        // 如果 item[key]有值
                        result.push(classNames(item[key]))
                    }
                }
            }

        })
    }

    // 输出 字符串带空格 遍历去空格
    return result.filter(item => item).join(' ');
}







function processOrder(order) {
    if (!order) {
        return null
    }
    if (order.status !== 'pending') {
        return;
    }
    // if 层级
    if (Array.isArray(order.items) && order.items.length > 0) {
        // reduce 方法
        order.total = order.items.reduce((acc, cur) => {
            if (cur.price) {
                acc += cur.price * (cur.quantity || 1)
            }
            return acc
        }, 0)

        order.status = 'calculated'
        return order
    }
}

async function fetchUserList(ids) {
    const result = []
    const res = await Promise.allSettled(ids.map(id => fetch(`/api/user/${id}`)));
    // 返回值格式转换字符串
    res.forEach(item => {
        if (item?.status === 'fulfilled' && item?.value?.name) {
            result.push(item.value.name)
        }
    })
    return result
}
