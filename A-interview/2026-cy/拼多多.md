单线程安全
避免多线程同时修改 DOM 造成混乱、冲突。
非阻塞
异步任务（定时器、ajax、事件）不会阻塞主线程，主线程可以继续执行代码、响应用户。
执行顺序可控
微任务优先于渲染，渲染优先于下一个宏任务，保证 UI 流畅、交互优先。

实现一个，自己的forEach 函数，第二个参数可以传递公共对象
this 指向

Array.prototype.myForEach = function (callback, thisArg) {
  // 1. 获取当前数组
  const arr = this;
  // 2. 判断是否为数组
  if (!Array.isArray(arr)) {
    throw new TypeError('myForEach must be called on array');
  }
  // 3. 判断回调是否是函数
  if (typeof callback !== 'function') {
    throw new TypeError(callback + ' is not a function');
  }

  const len = arr.length;
  let i = 0;

  // 4. 遍历（稀疏数组跳过空项，和原生一致）
  while (i < len) {
    if (arr.hasOwnProperty(i)) {
      // 重点：用 call 绑定 thisArg，传参：item, index, 原数组
      callback.call(thisArg, arr[i], i, arr);
    }
    i++;
  }
};




// 单任务队列
function fn(callback) {
    let nextTask = null;
    let isExecuting = false;

    return async (name) => {
        // 当前没有执行中的任务，也没有下一个任务
        if (!nextTask && !isExecuting) {
            isExecuting = true;
            await callback(name);
            isExecuting = false;
            // 如果 nextTask 存在，说明有任务在执行中，需要等待下一个任务执行完成
            if (nextTask) {
                isExecuting = true;
                await callback(nextTask);
                nextTask = null;
                isExecuting = false;
            }
        } else if (isExecuting) {
            nextTask = name;
        }
    }
}

const testFn = async (name) => {
    //生成1-10之间的一个随机数

    await new Promise((resolve) => {

    console.log("testFn =========>>", name);
